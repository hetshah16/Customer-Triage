"""
Loads all markdown files from data/ corpus into ChromaDB.
Exposes retrieve(query, company) -> list of relevant text chunks.
"""
import os
import glob
import chromadb
from chromadb.utils import embedding_functions

# Deterministic seed for reproducibility
COLLECTION_NAME = "support_corpus"
DATA_ROOT = os.path.join(os.path.dirname(__file__), "..", "data")

# Map company names to corpus folders
COMPANY_TO_FOLDER = {
    "HackerRank": "hackerrank",
    "Claude":     "claude",
    "Visa":       "visa",
    "None":       None  # search all
}

def _get_collection():
    """Init ChromaDB in-memory client with sentence-transformer embeddings."""
    client = chromadb.Client()  # in-memory, no persistence needed
    ef = embedding_functions.SentenceTransformerEmbeddingFunction(
        model_name="all-MiniLM-L6-v2"  # fast, free, local
    )
    return client, ef

def build_retriever():
    """Load all .md files → chunk → embed → store in ChromaDB. Returns collection."""
    print("[RETRIEVER_INIT] -> Loading corpus from data/...")
    client, ef = _get_collection()

    # Drop and recreate for determinism
    try:
        client.delete_collection(COLLECTION_NAME)
    except Exception:
        pass

    collection = client.create_collection(
        name=COLLECTION_NAME,
        embedding_function=ef
    )

    docs, ids, metas = [], [], []
    all_md = glob.glob(os.path.join(DATA_ROOT, "**", "*.md"), recursive=True)

    for i, filepath in enumerate(all_md):
        with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
            content = f.read().strip()
        if not content:
            continue

        # Derive company from path
        rel = os.path.relpath(filepath, DATA_ROOT)
        company = rel.split(os.sep)[0]  # hackerrank / claude / visa

        # Chunk: split by double newline, max 500 chars
        chunks = [c.strip() for c in content.split("\n\n") if len(c.strip()) > 30]
        for j, chunk in enumerate(chunks[:10]):  # max 10 chunks per file
            docs.append(chunk[:500])
            ids.append(f"{i}_{j}")
            metas.append({"company": company, "source": filepath})

    collection.add(documents=docs, ids=ids, metadatas=metas)
    print(f"[RETRIEVER_INIT] -> Indexed {len(docs)} chunks from {len(all_md)} files.")
    return collection

def retrieve(collection, query: str, company: str = None, n=3) -> list[str]:
    """Query corpus. Filter by company if provided. Returns top-n text chunks."""
    print(f"[RETRIEVER] -> Querying: '{query[:60]}' | company={company}")

    where = None
    if company and company != "None":
        folder = COMPANY_TO_FOLDER.get(company)
        if folder:
            where = {"company": folder}

    results = collection.query(
        query_texts=[query],
        n_results=n,
        where=where
    )
    chunks = results["documents"][0] if results["documents"] else []
    print(f"[RETRIEVER] -> Found {len(chunks)} chunks.")
    return chunks

"""
Classifies ticket: product_area, request_type, urgency, should_escalate.
Uses Groq LLM with fallback.
"""
import json
from langchain_core.messages import SystemMessage, HumanMessage
from prompts import CLASSIFIER_PROMPT

# Keywords that force escalation regardless of LLM output
ESCALATE_KEYWORDS = [
    "stolen", "fraud", "unauthorized", "breach", "hacked",
    "site is down", "not accessible", "billing dispute",
    "legal", "lawsuit", "locked out", "compromised"
]

def classify(llm, issue: str, subject: str, company: str) -> dict:
    """Classify ticket. Returns dict with product_area, request_type, urgency, should_escalate."""
    print(f"[CLASSIFIER] -> Classifying | company={company} | issue={issue[:60]}")

    # Hard escalation check before LLM call (saves tokens)
    issue_lower = issue.lower()
    force_escalate = any(kw in issue_lower for kw in ESCALATE_KEYWORDS)

    prompt = f"Company: {company}\nSubject: {subject}\nIssue: {issue}"
    try:
        response = llm.invoke([
            SystemMessage(content=CLASSIFIER_PROMPT),
            HumanMessage(content=prompt)
        ])
        result = json.loads(response.content.strip())
    except Exception as e:
        print(f"[CLASSIFIER_ERR] -> {e}")
        result = {
            "product_area": "general_support",
            "request_type": "product_issue",
            "urgency": "medium",
            "should_escalate": False
        }

    if force_escalate:
        result["should_escalate"] = True
        print(f"[CLASSIFIER] -> Force escalate triggered by keyword match.")

    print(f"[CLASSIFIER] -> Result: {result}")
    return result

"""
Generates final user-facing response or escalation message.
Grounded strictly in corpus chunks.
"""
import json
from langchain_core.messages import SystemMessage, HumanMessage
from prompts import RESPONDER_PROMPT

ESCALATION_RESPONSE = (
    "This issue requires immediate attention from our support team. "
    "A human agent will contact you shortly."
)

def respond(llm, issue: str, company: str, corpus_chunks: list, should_escalate: bool) -> dict:
    """Generate response. Returns dict: status, response, justification."""
    print(f"[RESPONDER] -> Generating response | escalate={should_escalate}")

    if should_escalate:
        print("[RESPONDER] -> Escalating ticket.")
        return {
            "status": "escalated",
            "response": ESCALATION_RESPONSE,
            "justification": "Ticket flagged as high-risk or sensitive. Routed to human agent."
        }

    corpus_text = "\n---\n".join(corpus_chunks) if corpus_chunks else "No relevant corpus found."
    prompt = RESPONDER_PROMPT.format(
        corpus=corpus_text,
        company=company,
        issue=issue
    )

    try:
        response = llm.invoke([
            SystemMessage(content="You are a grounded support agent. Return JSON only."),
            HumanMessage(content=prompt)
        ])
        # Strip markdown fences if present
        raw = response.content.strip().replace("```json", "").replace("```", "").strip()
        result = json.loads(raw)
    except Exception as e:
        print(f"[RESPONDER_ERR] -> {e}")
        result = {
            "status": "replied",
            "response": "I was unable to find relevant information in our support corpus for this issue.",
            "justification": "LLM parse error; defaulting to safe fallback response."
        }

    print(f"[RESPONDER] -> Status: {result.get('status')}")
    return result

"""
Entry point. Reads support_tickets.csv → runs triage agent → writes output.csv.
Deterministic: fixed model temperature=0.
"""
import os
import pandas as pd
from dotenv import load_dotenv
load_dotenv()

from retriever import build_retriever, retrieve
from classifier import classify
from responder import respond
from models import get_model, get_fallback_model
from groq import RateLimitError
import httpx

# Paths (relative to repo root)
INPUT_CSV  = os.path.join(os.path.dirname(__file__), "..", "support_tickets", "support_tickets.csv")
OUTPUT_CSV = os.path.join(os.path.dirname(__file__), "..", "support_tickets", "output.csv")

def get_llm():
    """Return LLM with fallback logic."""
    try:
        return get_model([])  # no tools needed for this task
    except Exception:
        return get_fallback_model([])

def process_ticket(llm, collection, row) -> dict:
    """Full pipeline for one ticket row."""
    issue   = str(row.get("Issue", "") or "")
    subject = str(row.get("Subject", "") or "")
    company = str(row.get("Company", "None") or "None")

    # Step 1: Classify
    classification = classify(llm, issue, subject, company)

    # Step 2: Retrieve corpus chunks
    query  = f"{subject} {issue}".strip()
    chunks = retrieve(collection, query, company, n=3)

    # Step 3: Generate response
    result = respond(llm, issue, company, chunks, classification["should_escalate"])

    return {
        "status":       result["status"],
        "product_area": classification["product_area"],
        "response":     result["response"],
        "justification":result["justification"],
        "request_type": classification["request_type"]
    }

def main():
    print("[MAIN] -> Starting support triage agent.")

    # Build corpus index once
    collection = build_retriever()

    # Init LLM
    llm = get_llm()

    # Load tickets
    df = pd.read_csv(INPUT_CSV)
    print(f"[MAIN] -> Loaded {len(df)} tickets from {INPUT_CSV}")

    results = []
    for i, row in df.iterrows():
        print(f"\n[MAIN] -> Processing ticket {i+1}/{len(df)}")
        try:
            out = process_ticket(llm, collection, row)
        except (RateLimitError, httpx.TimeoutException) as e:
            print(f"[MAIN] -> Model switch triggered: {e}")
            llm = get_fallback_model([])
            out = process_ticket(llm, collection, row)
        except Exception as e:
            print(f"[MAIN_ERR] -> Ticket {i} failed: {e}")
            out = {
                "status": "escalated",
                "product_area": "general_support",
                "response": "Unable to process this ticket. Escalating to human agent.",
                "justification": f"Processing error: {str(e)[:100]}",
                "request_type": "product_issue"
            }
        results.append(out)

    # Write output
    out_df = pd.DataFrame(results)
    out_df.to_csv(OUTPUT_CSV, index=False)
    print(f"\n[MAIN] -> Done. Output written to {OUTPUT_CSV}")
    print(out_df.head())

if __name__ == "__main__":
    main()
